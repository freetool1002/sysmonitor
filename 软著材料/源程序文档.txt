  321      // 累加所有活动网卡（或指定网卡）的收发字节总数
  322      private void AddNetTotals(ref long recv, ref long sent)
  323      {
  324          foreach (var n in NetworkInterface.GetAllNetworkInterfaces())
  325          {
  326              // 只统计"已启用"且"非回环"的网卡
  327              if (n.OperationalStatus != OperationalStatus.Up || n.NetworkInterfaceType == NetworkInterfaceType.Loopback)
  328                  continue;
  329              // 指定了网卡时，跳过其它网卡（空串=全部网络）
  330              if (adapterId != "" && n.Id != adapterId) continue;
  331              var s = n.GetIPv4Statistics();
  332              recv += s.BytesReceived;
  333              sent += s.BytesSent;
  334          }
  335      }
